cmake_minimum_required(VERSION 3.16)
project(OrbitSimLite VERSION 3.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CTest)
option(ORBITSIMLITE_BUILD_GUI "Build the SFML/OpenGL editor" ON)
option(ORBITSIMLITE_BUILD_DEMO "Build the application launchers" ON)
option(ORBITSIMLITE_BUILD_TESTS "Build numerical and workflow tests" ON)
option(ORBITSIMLITE_BUILD_BENCHMARKS "Build numerical benchmark" ON)
# Core owns numerical/document/file services and can be built without any graphics dependency.
add_library(orbitsimlite_core STATIC
    src/vec2.cpp src/body.cpp src/physics.cpp src/simulator.cpp
    src/trajectory_prediction.cpp src/scene.cpp src/scene_document.cpp
    src/simulation_controller.cpp src/editor_controller.cpp src/app_paths.cpp
    src/scene_serializer.cpp src/snapshot_exporter.cpp src/headless_runner.cpp src/scene_catalog.cpp)
target_include_directories(orbitsimlite_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(orbitsimlite_core SYSTEM PRIVATE third_party)
if(ORBITSIMLITE_BUILD_GUI)
    find_package(SFML 3 QUIET COMPONENTS Graphics Window System)
    if(SFML_FOUND)
        set(SFML_LIBS SFML::Graphics SFML::Window SFML::System)
    else()
        find_package(SFML 2.5 REQUIRED COMPONENTS graphics window system)
        set(SFML_LIBS sfml-graphics sfml-window sfml-system)
    endif()
    find_package(OpenGL REQUIRED)
    # Presentation and orchestration depend on SFML/OpenGL; the numerical core above does not.
    add_library(orbitsimlite STATIC
        src/actions.cpp
        src/application.cpp src/application_events.cpp src/application_input.cpp
        src/application_actions.cpp src/application_scene.cpp src/application_diagnostics.cpp
        src/editor_ui.cpp src/editor_ui_text.cpp src/editor_ui_widgets.cpp
        src/editor_ui_input.cpp src/editor_ui_panel.cpp src/editor_ui_dialogs.cpp
        src/editor_ui_detail.hpp
        src/renderer3d.cpp src/renderer_draw.cpp src/renderer_field.cpp)
    target_link_libraries(orbitsimlite PUBLIC orbitsimlite_core ${SFML_LIBS} OpenGL::GL)
    target_include_directories(orbitsimlite SYSTEM PRIVATE third_party)
    target_compile_definitions(orbitsimlite PRIVATE ORBITSIMLITE_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
    file(COPY assets DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
else()
    add_library(orbitsimlite ALIAS orbitsimlite_core)
endif()
# Thin startup layer shared by all executable entry points; GUI selection is a compile-time option.
add_library(orbitsimlite_launcher STATIC src/app_launcher.cpp)
target_link_libraries(orbitsimlite_launcher PUBLIC orbitsimlite)
if(ORBITSIMLITE_BUILD_GUI)
    target_compile_definitions(orbitsimlite_launcher PRIVATE ORBITSIMLITE_HAS_GUI=1)
endif()
set(warning_targets orbitsimlite_core orbitsimlite_launcher)
if(ORBITSIMLITE_BUILD_GUI)
    list(APPEND warning_targets orbitsimlite)
endif()
foreach(target ${warning_targets})
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
if(ORBITSIMLITE_BUILD_DEMO)
    foreach(app orbit_simulator demo_solar_system demo_binary_stars demo_threebody_figure8 demo_simulator)
        add_executable(${app} examples/${app}.cpp)
        target_link_libraries(${app} PRIVATE orbitsimlite_launcher)
        install(TARGETS ${app} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    endforeach()
endif()
# All CTest suites link only the core so CI can exercise them without a desktop/display.
if(ORBITSIMLITE_BUILD_TESTS)
    foreach(suite physics nbody spatial product)
        add_executable(orbitsimlite_${suite}_tests tests/${suite}_tests.cpp)
        target_link_libraries(orbitsimlite_${suite}_tests PRIVATE orbitsimlite_core)
        add_test(NAME ${suite} COMMAND orbitsimlite_${suite}_tests)
    endforeach()
endif()
if(ORBITSIMLITE_BUILD_BENCHMARKS)
    add_executable(benchmark_integrators examples/benchmark_integrators.cpp)
    target_link_libraries(benchmark_integrators PRIVATE orbitsimlite_core)
endif()
install(TARGETS orbitsimlite_core orbitsimlite_launcher ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
if(ORBITSIMLITE_BUILD_GUI)
    install(TARGETS orbitsimlite ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY assets/ DESTINATION ${CMAKE_INSTALL_DATADIR}/orbitsimlite)
install(FILES third_party/nlohmann-LICENSE.MIT DESTINATION ${CMAKE_INSTALL_DATADIR}/orbitsimlite)
set(CPACK_GENERATOR "TGZ")
include(CPack)
